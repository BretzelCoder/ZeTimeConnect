#
#     pip install -r requirements-lock.txt

#

#
#     pip install --upgrade bleak
#     pip freeze | grep -i "bleak\|winrt\|typing_extensions"
#
# (un "pip freeze" nu recopierait tout l'environnement Python de la machine,
# pytest et le reste compris -- seul l'arbre de bleak a sa place ici.)
